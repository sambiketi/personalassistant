@router.get("/progress/{user_id}")
async def get_progress(user_id: str):
    """Get user progress"""
    try:
        user = db.get_or_create_user(user_id)
        agent = get_agent(user_id)
        response = agent.handle_show_progress()
        
        # Also get Vanguard stats if available
        vanguard_data = None
        try:
            if agent and hasattr(agent, 'get_vanguard_brain'):
                brain = agent.get_vanguard_brain(user_id)
                if brain:
                    vanguard_data = brain.get_stats()
        except Exception as e:
            logger.warning(f"Could not get Vanguard stats: {e}")
        
        return {
            "success": True,
            "progress": response.message if response else "No progress data available",
            "vanguard": vanguard_data
        }
    except Exception as e:
        logger.error(f"Error getting progress: {e}")
        # Return a graceful error response instead of 500
        return {
            "success": False,
            "progress": "Unable to load progress data. Please try again.",
            "vanguard": None
        }
